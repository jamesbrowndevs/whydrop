```sql
SELECT country,
       SUM(CASE WHEN month='2026-09' THEN revenue_usd ELSE 0 END) -
       SUM(CASE WHEN month='2026-08' THEN revenue_usd ELSE 0 END) AS rev_change_usd,
       SUM(CASE WHEN month='2026-09' THEN orders ELSE 0 END) -
       SUM(CASE WHEN month='2026-08' THEN orders ELSE 0 END) AS order_change,
       ROUND(
           (SUM(CASE WHEN month='2026-09' THEN revenue_usd ELSE 0 END) -
            SUM(CASE WHEN month='2026-08' THEN revenue_usd ELSE 0 END)) * 100.0 /
           NULLIF(
               (SELECT SUM(CASE WHEN month='2026-09' THEN revenue_usd ELSE 0 END) -
                      SUM(CASE WHEN month='2026-08' THEN revenue_usd ELSE 0 END)
               FROM (
                   SELECT date_format(to_date(date), 'yyyy-MM') AS month,
                          SUM(unit_price_usd * quantity) AS revenue_usd
                   FROM WHYDROP_FILES.orders
                   WHERE country IN ('US','GB','IN','CA','DE','AU')
                     AND date_format(to_date(date), 'yyyy-MM') IN ('2026-08','2026-09')
                   GROUP BY month
               )
           ), 1),
           2) AS pct_of_total_drop
FROM (
    SELECT o.country,
           date_format(to_date(o.date), 'yyyy-MM') AS month,
           SUM(o.unit_price_usd * o.quantity) AS revenue_usd,
           COUNT(*) AS orders
    FROM WHYDROP_FILES.orders o
    WHERE o.country IN ('US','GB','IN','CA','DE','AU')
      AND date_format(to_date(o.date), 'yyyy-MM') IN ('2026-08','2026-09')
    GROUP BY o.country, month
) sub
GROUP BY country
```
