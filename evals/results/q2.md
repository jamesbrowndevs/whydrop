<details><summary>Q15 [H3] Conversion rate (orders per session) for paid_social traffic by month - 2 rows</summary>

```sql
WITH orders_month AS (
    SELECT date_format(to_date(o.date), 'yyyy-MM') AS month,
           COUNT(*) AS total_orders
    FROM WHYDROP_FILES.orders o
    WHERE o.product_id = 'P001'
      AND o.channel = 'paid_social'
      AND o.date >= '2026-08-01'
      AND o.date < '2026-10-01'
    GROUP BY date_format(to_date(o.date), 'yyyy-MM')
),
traffic_month AS (
    SELECT date_format(to_date(w.date), 'yyyy-MM') AS month,
           SUM(CAST(w.sessions AS DOUBLE)) AS total_sessions
    FROM WHYDROP_FILES.web_traffic w
    WHERE w.source = 'paid_social'
      AND w.date >= '2026-08-01'
      AND w.date < '2026-10-01'
    GROUP BY date_format(to_date(w.date), 'yyyy-MM')
)
SELECT o.month,
       o.total_orders,
       t.total_sessions,
       CASE WHEN t.total_sessions = 0 THEN 0 ELSE o.total_orders / t.total_sessions END AS orders_per_session
FROM orders_month o
JOIN traffic_month t USING (month)
ORDER BY o.month
```
